🧠 MODEL ARCHITECTURE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Base Model:     ResNet50 (Pre-trained on ImageNet)
Input Size:     224 x 224 RGB images
Output:         120 dog breed classes
Transfer:       Fine-tuning with frozen early layers
Optimizer:      Adam with learning rate scheduling


📊 TRAINING CONFIGURATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Epochs:         20 (configurable)
Batch Size:     32 (reduce if GPU memory limited)
Learning Rate:  0.001 with plateau scheduling
Split:          80% training, 20% validation
Workers:        4 (parallel data loading)

🐛 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Problem: "No module named 'torch'"
Solution: pip install torch torchvision

Problem: "CUDA out of memory"
Solution: Reduce BATCH_SIZE in train.py or config.py

Problem: "Images directory not found"
Solution: Ensure images/Images/ folder exists with breed subfolders

Problem: Training is slow
Solution: Use GPU, increase NUM_WORKERS, or reduce image size
